```python
import os
import sys
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

from database.db import get_session
from database.models import User, Organisation
from werkzeug.security import generate_password_hash

EMAIL = 'andre-test@nerve.local'

        org_id=org.id,
```

```bash
if [ -f "$BOX_KEY" ] && [ -n "$BOX_USER" ] && [ "$BOX_USER" != "uXXXXX" ]; then
    echo "[BACKUP] Schicht 2: rsync Push → ${BOX_HOST}${BOX_PATH}/"
    rsync \
        --archive \
        --compress \
        -e "ssh -p23 -i ${BOX_KEY} -o StrictHostKeyChecking=no -o BatchMode=yes" \
        "$BACKUP_FILE" \
        "${BOX_USER}@${BOX_HOST}:${BOX_PATH}/"
    echo "[BACKUP] Schicht 2: Push OK"

    # 90d-Rotation: list remote → parse timestamp from filename → delete old files
    # Storage Box hat keine volle Shell — kein remote find/awk, nur SSH-Einzelbefehle
    CUTOFF_DATE=$(date -d "-${BOX_RETENTION_DAYS} days" +%Y-%m-%d)
    OLD_FILES=$(ssh -p23 -i "${BOX_KEY}" \
        "${BOX_USER}@${BOX_HOST}" \
        "ls ${BOX_PATH}/" 2>/dev/null \
        | grep '^nerve-' \
        | awk -F'[_.]' '{print $1"-"$2}' \
        | while read fname date_part; do
            [ "$date_part" \< "$CUTOFF_DATE" ] && echo "$fname"
          done)
    for old_file in $OLD_FILES; do
        ssh -p23 -i "${BOX_KEY}" "${BOX_USER}@${BOX_HOST}" \
            "rm ${BOX_PATH}/${old_file}" 2>/dev/null || true
        echo "[BACKUP] Schicht 2: Rotation geloescht: ${old_file}"
    done
else
    echo "[BACKUP] Schicht 2: BOX_KEY fehlt oder BOX_USER nicht konfiguriert — skip"
fi
```

**Critical rsync constraint from research:**
- Port 23 (NOT 22) for Hetzner Storage Box
- `-o BatchMode=yes` prevents interactive prompts in systemd context
- Rotation must be script-side — Storage Box has no shell for remote `find -mtime`
- Filename format `nerve-YYYY-MM-DD_HHMMSS.sql.gz` enables date-based rotation by parsing filename prefix

### `scripts/backup_training_to_s3.sh` — NEW IONOS S3 Object-Lock backup (utility, file-I/O)

**Analog:** `scripts/backup_postgres.sh` (copy structure: set -e, constants block, pg_dump|gzip, size check, upload, rotation, cleanup)

**Full target structure** (based on backup_postgres.sh pattern + RESEARCH.md Pattern 4 commands):
```bash
#!/bin/bash
# scripts/backup_training_to_s3.sh — NERVE Training Schema Backup (Schicht 3)
# Phase 08.23.2.D.UX.0
# Runs via systemd nerve-backup-training.timer (weekly)
# User: root (sources ionos-s3.env, connects as nerve_anon_worker via PGPASSWORD)
set -euo pipefail

SECRETS_FILE="/etc/nerve/ionos-s3.env"
if [ ! -f "$SECRETS_FILE" ]; then
    echo "[BACKUP-S3] FEHLER: $SECRETS_FILE nicht gefunden" >&2
    exit 1
fi
source "$SECRETS_FILE"

# ── Constants ────────────────────────────────────────────────────────────────
TIMESTAMP=$(date +%Y-%m-%d_%H%M%S)
BACKUP_KEY="training-${TIMESTAMP}.sql.gz"
TMP_FILE="/tmp/${BACKUP_KEY}"
MIN_SIZE_BYTES=512     # training schema starts empty — lower threshold than postgres backup
IONOS_ENDPOINT="${IONOS_S3_ENDPOINT}"  # from ionos-s3.env
BUCKET="${IONOS_S3_BUCKET}"            # from ionos-s3.env

echo "[BACKUP-S3] Start: $TIMESTAMP"

# ── pg_dump as nerve_anon_worker (has SELECT on training.*) ──────────────────
PGPASSWORD="$NERVE_ANON_WORKER_DB_PASSWORD" \
    pg_dump -U nerve_anon_worker -d nerve \
    --schema=training --no-owner --no-acl \
    | gzip > "$TMP_FILE"

# ── Size-Check ───────────────────────────────────────────────────────────────
ACTUAL_SIZE=$(stat -c%s "$TMP_FILE" 2>/dev/null || echo 0)
if [ "$ACTUAL_SIZE" -lt "$MIN_SIZE_BYTES" ]; then
    echo "[BACKUP-S3] FEHLER: Dump zu klein ($ACTUAL_SIZE bytes < $MIN_SIZE_BYTES)" >&2
    rm -f "$TMP_FILE"
    exit 1
fi

echo "[BACKUP-S3] pg_dump OK: $ACTUAL_SIZE bytes"

# ── Object-Lock Upload (COMPLIANCE, +30d retention) ─────────────────────────
# WICHTIG: aws s3 cp unterstützt kein --object-lock-* → muss aws s3api put-object sein
RETAIN_UNTIL=$(date -u -d '+30 days' +%Y-%m-%dT%H:%M:%SZ)
aws s3api put-object \
    --bucket "$BUCKET" \
    --key "$BACKUP_KEY" \
    --body "$TMP_FILE" \
    --object-lock-mode COMPLIANCE \
    --object-lock-retain-until-date "$RETAIN_UNTIL" \
    --endpoint-url "$IONOS_ENDPOINT"

echo "[BACKUP-S3] OK: s3://$BUCKET/$BACKUP_KEY (WORM until $RETAIN_UNTIL)"
rm -f "$TMP_FILE"

# ── 365d Rotation: delete objects whose WORM retention has expired ───────────
CUTOFF_DATE=$(date -u -d '-365 days' +%Y-%m-%d)
aws s3 ls "s3://${BUCKET}/" --endpoint-url "$IONOS_ENDPOINT" \
    | awk '{print $4}' \
    | grep '^training-' \
    | while read obj; do
        obj_date=$(echo "$obj" | grep -oP '\d{4}-\d{2}-\d{2}')
        if [ -n "$obj_date" ] && [ "$obj_date" \< "$CUTOFF_DATE" ]; then
            aws s3api delete-object \
                --bucket "$BUCKET" \
                --key "$obj" \
                --endpoint-url "$IONOS_ENDPOINT" 2>/dev/null && \
            echo "[BACKUP-S3] Rotation: $obj geloescht (>365d)"
        fi
      done

echo "[BACKUP-S3] Abgeschlossen: $(date)"
```

3. **training.transcript_archive must NOT appear in `database/models.py`** — Alembic autogenerate would try to manage it and potentially generate DROP migrations. All training.* DDL lives only in migration SQL.

5. **aws s3 cp cannot be used for Object-Lock** — only `aws s3api put-object` supports `--object-lock-mode COMPLIANCE`. This is a hard constraint, not a preference.

6. **rsync port 23** — Hetzner Storage Box uses SSH port 23 (not 22). All rsync/ssh commands to the Box must use `-p23` / `-e "ssh -p23 ..."`.

7. **inspect.sh cannot verify training.transcript_archive directly** — `inspect.sh schema transcript_archive` uses `\d transcript_archive` without schema qualifier, returns empty.
