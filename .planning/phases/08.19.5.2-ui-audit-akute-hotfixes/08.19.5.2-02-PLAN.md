```javascript
async function submitWizard() {
  const nextBtn = document.getElementById('wiz-next-btn');
  nextBtn.textContent = 'Erstelle…';
  nextBtn.disabled = true;

  const payload = {
    firma:               document.getElementById('w-name').value.trim(),
    branche:             document.getElementById('w-branche').value.trim(),
    produkt:             document.getElementById('w-produkt').value.trim(),
    preismodell:         document.getElementById('w-preis').value.trim(),
    zielkunden:          document.getElementById('w-zielkunden').value.trim(),
    unternehmensgroesse: document.getElementById('w-groesse').value || null,
    einwaende:           getObjRows(),
    phasen:              getPhaseRows(),
  };

  try {
    const res = await fetch('/profiles/wizard', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'X-CSRFToken': _getCsrfToken(),
      },
      body: JSON.stringify(payload),
    });
    const d = await res.json();
    if (d.ok) {
      window.location.href = d.redirect;
    } else {
      alert(d.error || 'Fehler beim Erstellen');
      nextBtn.textContent = 'Profil erstellen ✓';
      nextBtn.disabled = false;
    }
  } catch(e) {
    alert('Netzwerkfehler');
    nextBtn.textContent = 'Profil erstellen ✓';
    nextBtn.disabled = false;
  }
}
```
