```python
from database.models import TranscriptSegment

# sp_map: int 0/1 -> role. Source-of-truth = live_session.py Z.886-887. Mirror it here.
_sp_map = {0: 'berater', 1: 'kunde'}

def _ts_to_ms(ts_str):
    # ts is 'HH:MM:SS' relative to call start -> ms offset
    try:
        h, m, s = (int(x) for x in str(ts_str).split(':'))
        return (h * 3600 + m * 60 + s) * 1000
    except Exception:
        return 0


def _transcript_entries_to_segments(log_entries):
    """Pure transform (WARN-3 testable helper): RAM entries -> list of segment dicts.
    Filters non-transcript + empty-text entries, maps speaker int->role (unknown -> 'system'),
    derives ts_ms. Returns [{'conversation_log_id'?:..., 'ts_ms':int, 'speaker':str, 'text':str}].
    Ordering = list order (authoritative). ts_ms is best-effort ms-from-start, monotonic-non-decreasing.
    """
    out = []
    for _entry in (log_entries or []):
        if not isinstance(_entry, dict) or _entry.get('type') != 'transcript':
            continue
        _txt = _entry.get('text') or ''
        if not _txt:
            continue
        out.append({
            'ts_ms': _ts_to_ms(_entry.get('ts')),
            'speaker': _sp_map.get(_entry.get('speaker'), 'system'),  # unknown int -> 'system' (CHECK-safe)
            'text': _txt,
        })
    return out

# --- inside api_beenden, after saved_conv_id = conv.id ---
# Idempotency / reentrance guard: skip if segments already exist for this conv.
_existing_seg = (db_conv.query(TranscriptSegment.id)
                 .filter(TranscriptSegment.conversation_log_id == conv.id)
                 .first())
if _existing_seg is None:
    _segs = _transcript_entries_to_segments(log_entries)
    for _s in _segs:
        db_conv.add(TranscriptSegment(
            conversation_log_id=conv.id,
            ts_ms=_s['ts_ms'],
            speaker=_s['speaker'],
            text=_s['text'],
        ))
    if _segs:
        db_conv.commit()
    print(f'[D.UX.1] transcript_segments INSERT conv={conv.id} added={len(_segs)}')
else:
    print(f'[D.UX.1] transcript_segments INSERT skipped (idempotent) conv={conv.id}')
```

After the synchronous transcript_segments INSERT (Task 1), spawn a daemon background thread that
double-writes the SAME transcript entries into `training.transcript_archive` (DA-06). This is

Key requirements:
- `source_call_hash`: a one-way hash that does NOT trivially reverse to the conversation_log id.
  Use `hashlib.sha256` over a server-secret-salted call identifier (e.g. f"{SECRET_KEY}:{conv.id}").
  Do NOT store the raw conv.id (anonymity). Use the same hash for all segments of one call.
- `ts_offset_ms`: per-CALL cryptographically random offset (Timestamp-Korrelations-Angriff-Schutz,
  Gemini 2026-05-28). Generate ONCE per call: `import secrets; _off = secrets.randbelow(86_400_000)`
  (0..24h). Store `ts_ms + _off` (or the per-segment ts plus the constant per-call offset) so absolute
  timing cannot be recovered via calls.started_at. Same offset for every segment of the call.
- `segment_index`: enumerate(0..n) over the transcript entries (ordering preserved without absolute ts).
- `speaker` / `text`: same mapped values as Task 1 (reuse `_transcript_entries_to_segments`).
- `schema_version`: 1.

Pattern:
```python
import threading, hashlib, secrets
from config import SECRET_KEY  # or current app secret source used elsewhere

_flask_app = current_app._get_current_object()
_seg_payload = [{'speaker': _s['speaker'], 'text': _s['text']}
                for _s in _transcript_entries_to_segments(log_entries)]
_conv_id_for_hash = conv.id

def _archive_bg(app, segs, conv_id):
    with app.app_context():
        _db = get_session()
        try:
            _hash = hashlib.sha256(f"{SECRET_KEY}:{conv_id}".encode('utf-8')).hexdigest()
            _off = secrets.randbelow(86_400_000)  # per-call random ms offset
            for _i, _s in enumerate(segs):
                _db.execute(text(
                    "INSERT INTO training.transcript_archive "
                    "(source_call_hash, segment_index, speaker, text, ts_offset_ms, schema_version) "
                    "VALUES (:h, :idx, :sp, :tx, :off, 1)"
                ), {'h': _hash, 'idx': _i, 'sp': _s['speaker'], 'tx': _s['text'], 'off': _off})
            _db.commit()
            print(f'[D.UX.1] training.transcript_archive bg-write hash={_hash[:8]} n={len(segs)}')
        except Exception as _e:
            # WARN-6: a MISSING training.transcript_archive (D.UX.0 not deployed) must be LOUD,
            # not silently swallowed alongside ordinary exceptions. Detect the undefined-table
            # condition and log a DISTINCT marker so it surfaces in inspect.sh logs-errors.
            _emsg = str(_e).lower()

                    ('does not exist' in _emsg or 'undefined' in _emsg or 'undefinedtable' in _emsg
                     or 'no such table' in _emsg or 'relation' in _emsg)):
                print('[D.UX.1] training.transcript_archive MISSING — D.UX.0 not deployed')
            else:
                print(f'[D.UX.1] training.transcript_archive bg-write Fehler: {_e}')
            _db.rollback()
        finally:
            _db.close()

if _seg_payload:
    threading.Thread(target=_archive_bg, args=(_flask_app, _seg_payload, _conv_id_for_hash),
                     daemon=True).start()
```
