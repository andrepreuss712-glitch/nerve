def pruefe_anwendung(vorher_fokus: dict, berater_texte, *, firmenname=None) -> dict | None:
    """Zaehlt das Fokus-Kriterium des Vor-Anrufs im Transkript des Folge-Anrufs nach.

    Args:
        vorher_fokus: der '_fokus'-Block des VORHERIGEN Anrufs
            ({'focus_key', 'count', 'katalog_version', ...}). None oder ohne focus_key -> None.
        berater_texte: die BERATER-Segment-Texte des AKTUELLEN Anrufs, EWB-Zeilen bereits
            gefiltert (services/transkript_renderer.segmente_ohne_ewb).
        firmenname: nur fuer die Firmenname-Teilregel; sonst None.

    Returns:
        None, wenn es keinen Vor-Fokus gibt (der Normalfall beim ersten Anruf).
        Sonst:
        {'focus_key': str,
         'vorher_count': int|None,      # was der Vor-Anruf gezaehlt hatte
         'jetzt_count': int,            # dieselbe Regel, auf dem neuen Transkript
         'angewendet': bool,            # die Regel feuert JETZT nicht mehr
         'katalog_version': int}

    ENTSCHEIDUNGSREGEL, bewusst einfach (Punkt 27): 'angewendet' ist True, wenn dieselbe Regel
    auf dem neuen Transkript NICHT MEHR ausloest. Sie ist damit dasselbe Ja/Nein, das der
    Katalog schon kennt — keine zweite Schwelle, keine zweite Wahrheit.
    ⚠ Bewusst NICHT "jetzt_count < vorher_count": eine Verbesserung von 9 auf 5 Nennungen von
    'we provide' bleibt eine Verletzung. Wer sie als Erfolg meldet, lobt fuer etwas, das der
    Katalog weiter beanstandet — genau die gefaellige Pruefung, die hier ausgeschlossen ist.

    KATALOG-VERSIONS-BRUCH: Traegt vorher_fokus eine andere katalog_version als der heutige
    Katalog, wird None zurueckgegeben. Sonst verglichen wir gegen eine Regel, die es so nicht
    mehr gibt — und meldeten "angewendet", weil die Regel verschwunden ist.
    """
