**scripts/backup_postgres.sh:**

```bash
#!/bin/bash
# Postgres Backup Script — NERVE (Phase 08.23.2.A)
# Runs via systemd nerve-backup.timer (daily at 03:30 UTC)
# User: postgres (peer auth, no password needed)
set -e

BACKUP_DIR="/opt/nerve/backups/postgres"
TIMESTAMP=$(date +%Y-%m-%d_%H%M%S)
BACKUP_FILE="$BACKUP_DIR/nerve-$TIMESTAMP.sql.gz"
MIN_SIZE_BYTES=1024  # < 1KB = silent pg_dump failure

mkdir -p "$BACKUP_DIR"

echo "[BACKUP] Starting pg_dump at $TIMESTAMP"
pg_dump nerve | gzip > "$BACKUP_FILE"

# Size check — catch silent pg_dump failures
ACTUAL_SIZE=$(stat -c%s "$BACKUP_FILE" 2>/dev/null || echo 0)
if [ "$ACTUAL_SIZE" -lt "$MIN_SIZE_BYTES" ]; then
    echo "[BACKUP] FEHLER: Backup-Datei zu klein ($ACTUAL_SIZE bytes < $MIN_SIZE_BYTES). Loeschen und Abbruch."
    rm -f "$BACKUP_FILE"
    exit 1
fi

echo "[BACKUP] OK: $BACKUP_FILE ($ACTUAL_SIZE bytes)"

# Rolling retention: delete backups older than 30 days
find "$BACKUP_DIR" -name "nerve-*.sql.gz" -mtime +30 -delete
echo "[BACKUP] Retention cleanup: Dateien aelter als 30 Tage geloescht"

echo "[BACKUP] Abgeschlossen: $(date)"
```

Make executable: `chmod +x scripts/backup_postgres.sh`

Create documentation with the exact content of the three systemd files that must be installed on the server. These are NOT created by this script (they go in /etc/systemd/system/ on the server — done manually in the cutover plan):

```
# /etc/systemd/system/nerve-backup.service
[Unit]
Description=NERVE Postgres Backup
OnFailure=nerve-backup-alert.service

[Service]
Type=oneshot
User=postgres
ExecStart=/opt/nerve/scripts/backup_postgres.sh

# /etc/systemd/system/nerve-backup.timer
[Unit]
Description=Daily NERVE Postgres Backup Timer

[Timer]
OnCalendar=*-*-* 03:30:00
Persistent=true
RandomizedDelaySec=15min

[Install]
WantedBy=timers.target

# /etc/systemd/system/nerve-backup-alert.service
[Unit]
Description=NERVE Backup Failure Alert

[Service]
Type=oneshot
ExecStart=/bin/sh -c 'echo "[BACKUP-ALERT] nerve-backup.service fehlgeschlagen — $(date)" | systemd-cat -t nerve-backup -p err'
```
