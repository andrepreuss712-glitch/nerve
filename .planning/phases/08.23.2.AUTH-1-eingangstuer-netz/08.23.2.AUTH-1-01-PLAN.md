    (function () {
      // Globaler CSRF-fetch-Wrapper (Fable-SCOPE-OVERRIDE 2026-07-04, A3):
      // injiziert X-CSRFToken bei state-changing Methoden automatisch aus dem Meta-Tag.
      // Fixt alle bestehenden token-losen Schreib-Fetches app-weit an EINER Stelle.
      // BEKANNTE, NICHT abgedeckte Umgehungen (bewusst dokumentiert):
      //  - fetch ueber eine aliasierte/Options-Variable, die der statische Guard-Scan nicht sieht
      //  - XMLHttpRequest und navigator.sendBeacon (der Wrapper patcht NUR window.fetch)
      var STATE_CHANGING = { POST: 1, PUT: 1, PATCH: 1, DELETE: 1 };
      function getCsrfToken() {
        var m = document.querySelector('meta[name="csrf-token"]');
        return m ? m.getAttribute('content') : '';
      }
      var _origFetch = window.fetch;
      window.fetch = function (input, init) {
        init = init || {};
        var method = (init.method || 'GET').toUpperCase();
        if (STATE_CHANGING[method]) {
          var headers = new Headers(init.headers || {});
          if (!headers.has('X-CSRFToken')) headers.set('X-CSRFToken', getCsrfToken());
          init.headers = headers;
        }
        return _origFetch.call(this, input, init);
      };
    })();
    </script>
