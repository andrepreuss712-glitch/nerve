    ```bash
    schilder)
        tbl="$2"; sch="${3:-}"
        _validate_table "$tbl"
        echo "── Schild: ${sch:+$sch.}$tbl ──────────────"
        _run_psql "SET search_path TO public, crm, training;
          SELECT n.nspname, obj_description(c.oid,'pg_class') AS schild
          FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace
          WHERE c.relname='$tbl' AND n.nspname IN ('public','crm','training');"
        echo "── Spalten-Schilder ──"
        _run_psql "SELECT a.attname, col_description(a.attrelid,a.attnum) AS schild
          FROM pg_attribute a JOIN pg_class c ON c.oid=a.attrelid
          JOIN pg_namespace n ON n.oid=c.relnamespace
          WHERE c.relname='$tbl' AND n.nspname IN ('public','crm','training')
          AND a.attnum>0 AND NOT a.attisdropped ORDER BY a.attnum;"
        echo "── Migrations-Historie (welche Migrationen die Tabelle beruehren) ──"
        grep -l "$tbl" alembic/versions/*.py 2>/dev/null | while read f; do
          echo "$f:"; git log --oneline -- "$f" 2>/dev/null | head -5; done
        ;;
    ```
