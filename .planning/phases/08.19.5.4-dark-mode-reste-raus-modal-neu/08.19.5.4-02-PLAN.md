**Ersetzen durch vollständige Implementierung:**

```javascript
  var _pendingNavUrl = null;

  // ── Nav-Guard: Click-Interceptor + Modal-Steuerung (Phase 08.19.5.4) ──────────
  // Fängt Klicks auf alle Nav-Links + Logout-Link ab wenn Call aktiv ist.
  // beforeunload bleibt als Fallback fuer Tab-Close / Browser-Zurueck / URL-Eingabe.

  function _nerveNavConfirm(confirm) {
    var overlay = document.getElementById('nerveNavModal');
    if (confirm === true) {
      // "Call beenden & verlassen" — Mic stoppen dann navigieren
      if (overlay) overlay.classList.remove('open');
      _nerveNavCleanupListeners();
      _stopMic();
      if (_pendingNavUrl) {
        window.location.href = _pendingNavUrl;
      }
      _pendingNavUrl = null;
    } else {
      // "Hier bleiben" — Modal schliessen, keine Navigation
      if (overlay) overlay.classList.remove('open');
      _nerveNavCleanupListeners();
      _pendingNavUrl = null;
    }
  }

  // ESC-Handler und Overlay-Klick-Handler (gespeichert fuer removeEventListener)
  var _nerveNavOnKeydown = null;

  function _nerveNavCleanupListeners() {
    if (_nerveNavOnKeydown) {
      document.removeEventListener('keydown', _nerveNavOnKeydown);
      _nerveNavOnKeydown = null;
    }
  }

  function _nerveNavOpenModal(targetUrl) {
    _pendingNavUrl = targetUrl;
    var overlay = document.getElementById('nerveNavModal');
    if (!overlay) return;
    overlay.classList.add('open');

    // ESC-Taste: schliesst Modal ohne Navigation (== "Hier bleiben")
    _nerveNavOnKeydown = function (e) {
      if (e.key === 'Escape') {
        e.preventDefault();
        _nerveNavConfirm(false);
      }
    };
    document.addEventListener('keydown', _nerveNavOnKeydown);

    // Overlay-Klick ausserhalb .n-modal-card: schliesst Modal ohne Navigation
    overlay.onclick = function (e) {
      if (e.target === overlay) {
        _nerveNavConfirm(false);
      }
    };
  }

  // Click-Interceptor: alle .n-nav-item + a.popup-item-logout abonnieren
  // Selector D-01 (SPEC gesperrt): '.n-nav-item, a.popup-item-logout'
  // Bedingung (SPEC gesperrt): state.micStarted === true
  document.querySelectorAll('.n-nav-item, a.popup-item-logout').forEach(function (el) {
    el.addEventListener('click', function (e) {
      if (state.micStarted !== true) return; // Kein aktiver Call — normale Navigation
      var href = el.getAttribute('href');
      if (!href || href === '#' || href.indexOf('javascript:') === 0) return; // Programmatic nav, kein Page-Load
      e.preventDefault();
      _nerveNavOpenModal(href);
    });
  });
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| JS → DOM-Navigation | pip-launcher.js liest href-Attribut aus angeklicktem Element und navigiert via window.location.href |
| Modal → pip-launcher | onclick-Attribute in base.html rufen _nerveNavConfirm() auf — Scope ist window-global innerhalb IIFE |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08.19.5.4-03 | Tampering | Click-Interceptor href-Read | accept | `el.getAttribute('href')` liest nur den DOM-Wert des vom Browser geparsten `<a>`-Tags. Kein User-Input verarbeitet. Navigation via `window.location.href = _pendingNavUrl` — href kommt aus dem statisch gerenderten Template, kein XSS-Risiko |
| T-08.19.5.4-04 | Elevation of Privilege | Modal onclick-Attribut | accept | `onclick="_nerveNavConfirm(false/true)"` — Funktion ist im IIFE-Scope von pip-launcher.js. Kein Angreifer-Input. Funktion ruft nur _stopMic() + window.location.href auf gültige interne URLs |
| T-08.19.5.4-05 | Denial of Service | ESC-Handler auf document | accept | Event-Listener wird via _nerveNavCleanupListeners() nach jedem Confirm/Dismiss entfernt. Kein Memory-Leak-Risiko |
</threat_model>
