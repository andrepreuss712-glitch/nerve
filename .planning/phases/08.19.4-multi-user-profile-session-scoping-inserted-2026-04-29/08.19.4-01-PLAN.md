```python
import logging as _logging
_logger = _logging.getLogger(__name__)

_per_sid_profile: dict = {}     # {sid: (name, daten)} — per D-01
_per_sid_lock = threading.Lock()

_session_state: dict = {}       # {sid: {key: value, ...}} — per D-02
_session_state_lock = threading.Lock()

def set_profile_for_sid(sid: str, name: str, daten: dict) -> None:
    """Set profile cache for an active WebSocket SID. Analog to _deepgram_sessions pattern."""
    with _per_sid_lock:
        _per_sid_profile[sid] = (name or '', daten if isinstance(daten, dict) else {})

def get_profile_for_sid(sid: str) -> tuple:
    """Returns (name, daten) for SID. Returns ('', {}) + warning for unknown SID."""
    with _per_sid_lock:
        result = _per_sid_profile.get(sid)
    if result is None:
        _logger.warning(f"[SID] get_profile_for_sid: unknown SID {sid}")
        return ('', {})
    return result

def init_session_state(sid: str, user_id: int, org_id: int, profile_id=None,
                       market: str = 'dach', language: str = 'de',
                       mode: str = 'meeting') -> None:
    """Initialize _session_state entry for a new WebSocket connection."""
    with _session_state_lock:
        _session_state[sid] = {
            'user_id': user_id,
            'org_id': org_id,
            'active_profile_id': profile_id,
            'kaufbereitschaft': 30,
            'active_sid': sid,
            'market': market,
            'language': language,
            'mode': mode,
            'conversation_log': [],
            'berater_words': 0,
            'kunde_words': 0,
            'roles_swapped': False,
            'covered_phases': set(),
            # Tier 3 — opportunistic (include here, same code operation)
            'kaufbereitschaft_verlauf': [],
            '_bof_count': 0,
            '_pending_speaker': None,
            '_confirmed_speaker': None,
            '_pending_since': None,
            '_log_last_sp': None,
        }

def pop_session_state(sid: str) -> None:
    """Remove all state for a disconnecting SID. Call in disconnect handler."""
    with _session_state_lock:
        _session_state.pop(sid, None)
    with _per_sid_lock:
        _per_sid_profile.pop(sid, None)
    drop_matcher(sid)
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| WebSocket connect → SID creation | Untrusted user connects, SID is assigned by Flask-SocketIO |
| SID → profile lookup | SID is the auth identity for profile data — must not be forgeable |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08.19.4-01 | Information Disclosure | `_per_sid_profile` dict | mitigate | Dict keyed by SID (Flask-SocketIO assigns SID at handshake, client cannot choose SID). No cross-SID reads possible when all lookups use the SID from request.sid. |
| T-08.19.4-02 | Information Disclosure | `_session_state` dict | mitigate | Same as T-01. SID is the isolation boundary. `pop_session_state(sid)` on disconnect prevents stale access. |
| T-08.19.4-03 | Denial of Service | `_session_state` unbounded growth | accept | Ping-timeout (~5s) fires disconnect → cleanup. Worst case: zombie SIDs accumulate for 5s only. Acceptable at 50-user EA scale. |
</threat_model>
