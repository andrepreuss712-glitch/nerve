Schritt 2 — JSON-Daten aller Profile extrahieren (Python-Script, kein sqlite3-CLI):
  python3 - <<'PYEOF'
  import sqlite3, json, collections

  db = sqlite3.connect(".planning/audits/profile-schema-realdata-2026-04-29/salesnerve.db")
  rows = db.execute("SELECT id, daten FROM profiles").fetchall()

  # Analyse-Strukturen
  top_level_keys = collections.Counter()
  basis_keys = collections.Counter()
  key_types = collections.defaultdict(set)  # key -> set of Python-type-names
  profiles_count = len(rows)

  for pid, daten_raw in rows:
      try:
          d = json.loads(daten_raw) if daten_raw else {}
      except Exception:
          d = {}
      for k, v in d.items():
          top_level_keys[k] += 1
          key_types[k].add(type(v).__name__)
      basis = d.get('basis', {})
      if isinstance(basis, dict):
          for k, v in basis.items():
              basis_keys[k] += 1

  db.close()

  # KEY-FINDINGS.md schreiben (kein Profil-Content, nur Keys + Types)
  lines = [
      "# KEY-FINDINGS: Production-Profil-Schema-Analyse",
      f"Datum: 2026-04-29 | Profile gesamt: {profiles_count}",
      "",
      "## Top-Level-Keys (alle Profile)",
      "| Key | Vorkommt in N Profilen | Python-Typen |",
      "|-----|------------------------|--------------|",
  ]
  for k, cnt in sorted(top_level_keys.items(), key=lambda x: -x[1]):
      types_str = ", ".join(sorted(key_types[k]))
      lines.append(f"| {k} | {cnt}/{profiles_count} | {types_str} |")

  lines += [
      "",
      "## basis.*-Keys (alle Profile)",
      "| Key | Vorkommt in N Profilen |",
      "|-----|------------------------|",
  ]
  for k, cnt in sorted(basis_keys.items(), key=lambda x: -x[1]):
      lines.append(f"| {k} | {cnt}/{profiles_count} |")

  # Drift-Checks: welche Profile haben einwaende/phasen NUR top-level, NUR basis.*, oder beides?
  einwaende_toplevel = 0
  einwaende_basis = 0
  phasen_toplevel = 0
  phasen_basis = 0
  branche_toplevel = 0
  fragen_key = 0

  db2 = sqlite3.connect(".planning/audits/profile-schema-realdata-2026-04-29/salesnerve.db")
  rows2 = db2.execute("SELECT id, daten FROM profiles").fetchall()
  for pid, daten_raw in rows2:
      try:
          d = json.loads(daten_raw) if daten_raw else {}
      except Exception:
          d = {}
      if 'einwaende' in d: einwaende_toplevel += 1
      basis = d.get('basis', {}) or {}
      if 'einwaende' in basis: einwaende_basis += 1
      if 'phasen' in d: phasen_toplevel += 1
      if 'phasen' in basis: phasen_basis += 1
      if 'branche' in d: branche_toplevel += 1
      if 'fragen' in d: fragen_key += 1
  db2.close()

  lines += [
      "",
      "## Drift-Key-Verteilung (fuer Migration-Entscheidung)",
      f"- einwaende top-level: {einwaende_toplevel}/{profiles_count}",
      f"- basis.einwaende: {einwaende_basis}/{profiles_count}",
      f"- phasen top-level: {phasen_toplevel}/{profiles_count}",
      f"- basis.phasen: {phasen_basis}/{profiles_count}",
      f"- branche top-level: {branche_toplevel}/{profiles_count}",
      f"- fragen key (soll entfernt werden): {fragen_key}/{profiles_count}",
      "",
      "## Ungemappte Keys (nicht in ProfileSchema v2)",
      "Keys die im Production-Dump vorhanden sind aber NICHT in ProfileSchema v2 gelistet:",
  ]

  known_schema_keys = {
      'schema_version', 'basis', 'zielgruppe', 'zielkunde', 'value', 'ki',
      'schmerzen', 'meta', 'kaufsignale', 'nogos', 'wettbewerber', 'uebergaenge',
      'techniken', 'einwaende_detail', 'einwaende', 'phasen',
      # transitional keys (werden in v3 Migration entfernt)
      'fragen', 'branche', 'pitch', 'erlaubnis', 'opener',
  }
  unmapped = [k for k in top_level_keys if k not in known_schema_keys]
  for k in sorted(unmapped):
      lines.append(f"- {k} (in {top_level_keys[k]}/{profiles_count} Profilen)")
  if not unmapped:
      lines.append("- (keine ungemappten Keys gefunden)")

  with open(".planning/audits/profile-schema-realdata-2026-04-29/KEY-FINDINGS.md", "w", encoding="utf-8") as f:
      f.write("\n".join(lines) + "\n")

  print(f"KEY-FINDINGS.md geschrieben. {profiles_count} Profile analysiert.")
