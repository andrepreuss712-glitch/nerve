**Schritt 4: Pencil-Click Event-Delegierung**

Nach dem bestehenden Unsicher-Dot-Click-Handler (Z.832-884), neuen Handler hinzufügen:

```javascript
// Phase 08.23.2.D.UX REQ-D.UX-12 — Pencil-Edit Click-Handler
document.addEventListener('click', function(ev) {
  var pencilBtn = ev.target.closest && ev.target.closest('.outcome-pencil-btn');
  if (!pencilBtn) return;
  ev.stopPropagation();
  ev.preventDefault();
  var callId = pencilBtn.getAttribute('data-call-id');
  if (!callId) return;
  var row = pencilBtn.closest('.db2-session-row');
  if (!row) return;

  // Toggle: schließen wenn bereits offen
  var existing = row.nextSibling;
  if (existing && existing.classList && existing.classList.contains('outcome-pencil-accordion')) {
    existing.parentNode.removeChild(existing);
    return;
  }

  // 7-Button-Accordion erstellen
  var acc = document.createElement('div');
  acc.className = 'outcome-pencil-accordion';
  acc.setAttribute('data-call-id', callId);

  var opts7 = [
    ['meeting_booked',     'Termin gebucht'],
    ['callback',           'Rückruf vereinbart'],
    ['send_info',          'Infos zusenden'],
    ['wrong_person',       'Falsche Person'],
    ['gatekeeper_blocked', 'Gatekeeper blockt'],
    ['no_interest',        'Kein Interesse'],
    ['contract_signed',    'Abschluss'],
  ];

  opts7.forEach(function(pair) {
    var btn = document.createElement('button');
    btn.type = 'button';
    btn.className = 'n-btn-ghost outcome-correction-btn';
    btn.textContent = pair[1];
    btn.style.minHeight = '44px';  // Touch-Target

    btn.addEventListener('click', function(e) {
      e.stopPropagation();
      var csrfMeta = document.querySelector('meta[name="csrf-token"]');
      var csrfToken = csrfMeta ? csrfMeta.getAttribute('content') : '';

      // Derive followup_intent (same mapping as pip-launcher.js)
      var followupMap = {
        meeting_booked: 'meeting', callback: 'callback',
        send_info: 'send_info', gatekeeper_blocked: 'retry_internal'
      };
      var followupIntent = followupMap[pair[0]] || 'none';

      fetch('/api/calls/' + encodeURIComponent(callId) + '/correct_outcome', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-CSRFToken': csrfToken,
        },
        body: JSON.stringify({ outcome: pair[0], followup_intent: followupIntent }),
      })
      .then(function(r) { return r.ok ? r.json() : null; })
      .then(function(json) {
        if (json && json.ok) {
          // Update Outcome-Label in row
          var outcomeSpan = row.querySelector('.db2-session-outcome');
          if (outcomeSpan) {
            // Keep unsicher-dot if present, update text
            var dot = outcomeSpan.querySelector('.outcome-unsicher-dot');
            var pencil = outcomeSpan.querySelector('.outcome-pencil-btn');
            outcomeSpan.textContent = _outcomeLabelDe(json.outcome || pair[0]);
            if (dot) outcomeSpan.appendChild(dot);    // Unsicher-Dot bleibt
            if (pencil) outcomeSpan.appendChild(pencil);  // Pencil bleibt
          }
          // Update Score-Label in row (final_score vom Server — per D-W4-05)
          // nerve.css tokens: --btn-primary-bg-from (#00D4AA), --page-text-secondary (#6B7280), --btn-danger-text (#f87171)
          // Note: original code used #F85149 which is not a nerve.css token; --btn-danger-text (#f87171) is the correct danger token
          if (json.final_score !== undefined) {
            var scoreSpan = row.querySelector('.db2-session-score');
            if (scoreSpan) {
              scoreSpan.textContent = String(json.final_score);
              scoreSpan.style.color = json.final_score >= 70
                ? 'var(--btn-primary-bg-from, #00D4AA)'
                : (json.final_score >= 40
                  ? 'var(--page-text-secondary, #6B7280)'
                  : 'var(--btn-danger-text, #f87171)');
            }
          }
          // Accordion entfernen
          acc.parentNode.removeChild(acc);
        }
      })
      .catch(function(e) {
        console.error('[PhaseD.UX] pencil-edit Fehler:', e);
        // Visible user feedback (consistent with Plan 05 error pattern — button text "Fehler — erneut versuchen")
        acc.innerHTML = '<div style="color:var(--btn-danger-text,#f87171);padding:8px 4px">'
          + 'Fehler — bitte erneut versuchen</div>';
      });
    });
    acc.appendChild(btn);
  });

  row.parentNode.insertBefore(acc, row.nextSibling);
});
```
