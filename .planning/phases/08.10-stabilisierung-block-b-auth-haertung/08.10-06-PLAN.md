**Migrations-Strategie mit sys.exit(1) STOPS-Behavior:**

In der bestehenden Migrations-Section von app.py (der try/except Block mit `ALTER TABLE users ADD COLUMN` Pattern), eine neue Migration hinzufügen:

```python
# H-21: oauth_id UNIQUE-Constraint (idempotent, mit STOPS-Behavior bei Duplicates)
try:
    with engine.connect() as db:
        from sqlalchemy import text as _text
        # 1. Duplicate-Check VOR Constraint-Anlage
        dup_row = db.execute(_text(
            "SELECT oauth_id FROM users WHERE oauth_id IS NOT NULL "
            "GROUP BY oauth_id HAVING COUNT(*) > 1 LIMIT 1"
        )).fetchone()
        if dup_row:
            # H-21: STOPS — Duplicate oauth_ids gefunden. Manuelle Bereinigung nötig vor Deploy.
            # Listet ALLE Duplicates in der Console für einfache Identifikation.
            all_dups = db.execute(_text(
                "SELECT oauth_id, COUNT(*) as cnt FROM users WHERE oauth_id IS NOT NULL "
                "GROUP BY oauth_id HAVING COUNT(*) > 1"
            )).fetchall()
            import sys as _sys
            print("[DB] FEHLER: Duplicate oauth_ids gefunden — NERVE stoppt:", file=_sys.stderr)
            for row in all_dups:
                print(f"  oauth_id={row[0]}  (count={row[1]})", file=_sys.stderr)
            print("[DB] Bitte oauth_id-Duplicates manuell bereinigen, dann neu starten.", file=_sys.stderr)
            _sys.exit(1)
        else:
            # 2. Idempotent: CREATE UNIQUE INDEX IF NOT EXISTS (SQLite-spezifisch)
            db.execute(_text(
                "CREATE UNIQUE INDEX IF NOT EXISTS uq_users_oauth_id "
                "ON users(oauth_id) WHERE oauth_id IS NOT NULL"
            ))
            db.commit()
            print("[DB] Migration: uq_users_oauth_id UNIQUE-Index OK")
except SystemExit:
    raise  # sys.exit() nicht abfangen
except Exception as _e:
    print(f"[DB] oauth_id UNIQUE-Index Migration-Fehler: {_e}")
```

**Wichtig:** `except SystemExit: raise` — sys.exit() wirft SystemExit, das NICHT von `except Exception` abgefangen werden darf. Sonst würde der STOPS-Behavior still ignoriert.

**Einfügestelle:** In der bestehenden _migrate()-Funktion oder im Migration-Block wo die anderen ALTER TABLE Statements sind. Falls app.py keinen zentralen _migrate()-Block hat, sondern einzelne try/except-Blöcke direkt nach dem Blueprint-Setup: Dort einfügen, nach den bestehenden Migrations.

**Wichtig:** `from sqlalchemy import text` ist möglicherweise bereits importiert. Beim Lesen prüfen. Falls schon `text` importiert: Alias `_text` verwenden oder direkt `text` nutzen — kein Duplicate-Import.

def confirm_email():
    """H-18: Email-Confirmation für Microsoft-OAuth-User. Phase 08.10."""
    from itsdangerous import URLSafeTimedSerializer, BadSignature, SignatureExpired
    from config import SECRET_KEY
    token = request.args.get('token', '')
    db = get_session()
    try:
        email = URLSafeTimedSerializer(SECRET_KEY, salt='nerve-email-confirm').loads(token, max_age=86400)

@auth_bp.route('/auth/resend-confirm', methods=['POST'])
# BEWUSST ohne @login_required — User ist auth'd via Session, aber Gate blockt login_required.

@limiter.limit("3 per 10 minutes")
def resend_confirm():
    """H-18 VARIANTE B: Resend Confirmation-Email für Microsoft-OAuth-User."""
    from itsdangerous import URLSafeTimedSerializer
    from config import SECRET_KEY
    from services.email_service import send_confirmation_email
    # User aus Session lesen (session ist verfügbar auch wenn login_required Gate blockt)
    user_id = session.get('user_id')
    if not user_id:
        return jsonify({'ok': False, 'error': 'not authenticated'}), 401
    db = get_session()
    try:
        user = db.get(User, user_id)
        if not user:
            return jsonify({'ok': False, 'error': 'user not found'}), 404
        if getattr(user, 'email_confirmed', True):
            return jsonify({'ok': False, 'error': 'email already confirmed'}), 400
        confirm_token = URLSafeTimedSerializer(SECRET_KEY, salt='nerve-email-confirm').dumps(user.email)
        confirm_url = url_for('auth.confirm_email', token=confirm_token, _external=True)
        send_confirmation_email(user.email, confirm_url, getattr(user, 'vorname', '') or '')
        return jsonify({'ok': True})
    except Exception as _e:
        print(f'[AUTH] resend_confirm error: {_e}')
        return jsonify({'ok': False, 'error': 'internal'}), 500
    finally:
        db.close()
