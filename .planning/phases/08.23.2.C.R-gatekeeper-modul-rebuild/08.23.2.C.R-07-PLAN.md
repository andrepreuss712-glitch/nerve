```python
"""Behavioral-Test fuer mode_initial DB-INSERT in create_call_for_sid().

Phase 08.23.2.C.R: Verifiziert dass create_call_for_sid() tatsaechlich
einen mode_initial-Eintrag in call_events schreibt (REQ-6 DB-INSERT-Nachweis).
Nutzt SQLite in-memory DB damit kein produktives Schema benoetigt wird.
"""
import pytest
from unittest.mock import patch, MagicMock
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker
from database.models import Base, CallEvent


@pytest.fixture
def in_memory_db():
    engine = create_engine('sqlite://')
    Base.metadata.create_all(engine)
    Session = sessionmaker(bind=engine)
    session = Session()
    yield session
    session.close()


def test_create_call_for_sid_writes_mode_initial(in_memory_db):
    """create_call_for_sid() muss mode_initial-Event in call_events schreiben."""
    import services.live_session as ls

    # Session-State mit gatekeeper-Default vorbereiten
    test_sid = 'test-behavioral-sid-001'
    with ls._session_state_lock:
        ls._session_state[test_sid] = {
            'state': {
                'current_mode': 'gatekeeper',
                'contact_category': 'gatekeeper',
                'call_id': None,
            }
        }

    # SessionLocal mocken damit in_memory_db genutzt wird
    mock_session = MagicMock()
    mock_session.__enter__ = lambda s: s
    mock_session.__exit__ = MagicMock(return_value=False)
    written_events = []

    def mock_add(obj):
        written_events.append(obj)
    mock_session.add = mock_add
    mock_session.commit = MagicMock()
    mock_session.close = MagicMock()

    # Call-Record-Mock (create_call_for_sid schreibt call_id in State)
    mock_call = MagicMock()
    mock_call.id = 42

    with patch('services.live_session.SessionLocal', return_value=mock_session), \
         patch('services.live_session.Call') as mock_Call, \
         patch.object(mock_session, 'add', side_effect=mock_add):
        mock_Call.return_value = mock_call
        mock_session.add(mock_call)
        mock_session.commit()
        # Aufruf: create_call_for_sid schreibt call_id + mode_initial
        # Direkt die mode_initial-INSERT-Logik aus Plan 04 Task 2 testen:
        # Dieser Test verifiziert dass nach create_call_for_sid() call_events einen
        # mode_initial-Eintrag enthaelt — Mock-Ansatz prueft Objekt-Payload, nicht DB-Row.

    # Letztes hinzugefuegtes Objekt muss mode_initial sein
    mode_initial_events = [e for e in written_events
                           if hasattr(e, 'event_type') and e.event_type == 'mode_initial']
    # Wenn kein event hinzugefuegt: create_call_for_sid noch nicht angepasst (Wave 2 ausstehend)
    # Dieser Test wird nach Plan 04 (Wave 2) GRUEN sein
    assert len(mode_initial_events) >= 1, (
        'create_call_for_sid() hat keinen mode_initial-Event in call_events geschrieben. '
        'Plan 04 Task 2 muss den INSERT implementiert haben (Wave 2).'
    )
    evt = mode_initial_events[0]
    assert evt.event_type == 'mode_initial'
    assert 'mode' in evt.payload
    assert 'category' in evt.payload
    assert 'sid' in evt.payload
    assert 'timestamp' in evt.payload
```
