  function _toIsoWithOffset(localVal) {
    // localVal = "2026-06-03T10:00" (naiv, Browser-Lokalzeit).
    if (!localVal) return null;
    var d = new Date(localVal);            // interpretiert als LOKALE Zeit des Browsers
    if (isNaN(d.getTime())) return null;
    var off = -d.getTimezoneOffset();      // Minuten; z.B. +120 fuer MESZ
    var sign = off >= 0 ? '+' : '-';
    var abs = Math.abs(off);
    var hh = String(Math.floor(abs / 60)).padStart(2, '0');
    var mm = String(abs % 60).padStart(2, '0');
    function p(n){ return String(n).padStart(2, '0'); }
    // Lokale Wandzeit + expliziter Offset -> unmissverstaendlicher Instant fuer das Backend.
    return d.getFullYear() + '-' + p(d.getMonth()+1) + '-' + p(d.getDate())
         + 'T' + p(d.getHours()) + ':' + p(d.getMinutes()) + ':' + p(d.getSeconds())
         + sign + hh + ':' + mm;            // z.B. "2026-06-03T10:00:00+02:00"
  }
