```python
# D-06: Du/Sie detection heuristic (2-trigger threshold)
_DU_FORMS = [
    'kannst du', 'hast du', 'bist du', 'machst du', 'willst du',
    'weisst du', 'weisst du', 'denkst du', 'sagst du', 'hoerst du',
    'siehst du', 'brauchst du', 'kennst du', 'findest du', 'glaubst du',
]
# Note: CLAUDE.md ASCII rule — variable names use 'oe', 'ae' etc. String content uses real Umlauts.

def _check_anrede_switch(sio, sid, transcript_text: str, ls_module) -> None:
    """Check if transcript contains Du-forms indicating anrede switch. Non-blocking."""
    try:
        text_lower = transcript_text.lower()
        _du_count = sum(1 for form in _DU_FORMS if form in text_lower)
        if _du_count >= 2:
            # Only trigger if current anrede is 'sie' (avoid duplicate events)
            _current_anrede = 'sie'
            try:
                with ls_module._session_state_lock:
                    _current_anrede = ls_module._session_state.get(sid, {}).get('session_anrede', 'sie')
            except Exception:
                pass
            if _current_anrede == 'sie':
                sio.emit('anrede_switch_detected', {
                    'detected_form': 'du',
                    'confidence': round(min(1.0, _du_count / 3.0), 2),
                    'sid': sid,
                }, room=sid)
                print(f"[Anrede-Detection] Du-switch detected: {_du_count} forms in utterance sid={sid}")
    except Exception as _ae:
        print(f"[Anrede-Detection] check failed (non-fatal): {_ae}")
```
