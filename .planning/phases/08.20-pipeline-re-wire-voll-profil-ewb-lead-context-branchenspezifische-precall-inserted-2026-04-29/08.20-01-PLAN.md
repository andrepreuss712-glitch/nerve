```python
def set_briefing_for_sid(sid: str, briefing_text: str) -> None:
    """Cache PreCall-Briefing text. Ghost-SID guard: drops silently if SID no longer active.
    Uses _session_state_lock — TOCTOU-safe (check and write are atomic under same lock)."""
```

## HIGH-1 Fix: Ghost-SID Guard on set_profile_for_sid()

Current `set_profile_for_sid()` (line 213-216) does NOT check if SID is active.
It must be updated to guard against Ghost SIDs:

```python
def set_profile_for_sid(sid: str, name: str, daten: dict) -> None:
    """Cache profile for an active WebSocket SID. Ghost-SID guard prevents leak if
    async load completes after disconnect (TOCTOU-safe under _per_sid_lock)."""
    with _per_sid_lock:
        # Ghost-SID guard: only write if session state confirms active connection.
        # Check _session_state (separate lock not needed here — per_sid_lock is sufficient
        # for per_sid_profile; _session_state is a secondary existence check without lock
        # because we only need approximate liveness, not strict consistency here).
        # For strict atomic check: verify SID in _session_state before writing.
        _session_state[sid]  # raises KeyError if not present — catch it
        _per_sid_profile[sid] = (name or '', daten if isinstance(daten, dict) else {})
```

Wait — the above approach requires holding two locks simultaneously. Correct Ghost-SID
guard for set_profile_for_sid uses a try/except pattern without needing _session_state_lock:

```python
def set_profile_for_sid(sid: str, name: str, daten: dict) -> None:
    """Cache profile for an active WebSocket SID. Ghost-SID guard: if SID is not in
    _session_state, drop the write (async race condition prevention — HIGH-1 fix)."""
    with _per_sid_lock:
        # Ghost-SID check: SID must be active. Reading _session_state without its lock
        # is safe here because: (a) dict read in CPython is GIL-protected, (b) we only
        # need 'exists' check — a concurrent pop() may remove it after this check but
        # before the write, which is acceptable (rare race, data discarded on next pop).
        if sid not in _session_state:
            _logger.debug(f"[SID] set_profile_for_sid: Ghost SID {sid} — dropping (async race)")
            return
        _per_sid_profile[sid] = (name or '', daten if isinstance(daten, dict) else {})
```

Note: `_per_sid_transcript` has NO setter function — it's written directly via `.setdefault()`
in `_flush_segment()`. No Ghost-SID guard needed there (setdefault is safe — worst case
creates an empty entry that gets cleaned by the next pop_session_state).

```python
def test_deadlock_stress_no_deadlock():
    """HIGH-2: 10 parallel threads doing Connect->write briefing->Disconnect must complete
    without deadlock within 10 seconds."""
    import services.live_session as ls
    import threading
    import time

    NUM_THREADS = 10
    TIMEOUT_S = 10
    results = []
    barrier = threading.Barrier(NUM_THREADS)

    def session_lifecycle(thread_id):
        sid = f'stress-test-sid-{thread_id:03d}'
        barrier.wait()  # Sync all threads to start simultaneously
        try:
            # Simulate Connect: init session state
            with ls._session_state_lock:
                ls._session_state[sid] = {'user_id': thread_id, 'org_id': 1}
            # Simulate PreCall briefing write
            ls.set_briefing_for_sid(sid, f'Briefing for thread {thread_id}')
            # Verify briefing is readable
            result = ls.get_briefing_for_sid(sid)
            assert result == f'Briefing for thread {thread_id}', f'Briefing mismatch: {result}'
            # Simulate Disconnect
            ls.pop_session_state(sid)
            # Verify cleanup
            with ls._session_state_lock:
                assert sid not in ls._session_state, f'SID {sid} not cleaned after pop'
            results.append(('ok', thread_id))
        except Exception as e:
            results.append(('error', thread_id, str(e)))

    threads = [threading.Thread(target=session_lifecycle, args=(i,)) for i in range(NUM_THREADS)]
    for t in threads:
        t.start()

    deadline = time.monotonic() + TIMEOUT_S
    for t in threads:
        remaining = deadline - time.monotonic()
        t.join(timeout=max(0, remaining))

    errors = [r for r in results if r[0] == 'error']
    not_done = NUM_THREADS - len(results)
    assert not_done == 0, f'{not_done} threads did not complete within {TIMEOUT_S}s (deadlock?)'
    assert errors == [], f'Thread errors: {errors}'
    print(f'test_deadlock_stress_no_deadlock: PASS ({NUM_THREADS} threads, no deadlock)')
```
