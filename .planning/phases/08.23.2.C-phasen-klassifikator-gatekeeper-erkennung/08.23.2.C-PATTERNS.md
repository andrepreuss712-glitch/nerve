# Source: services/anonymization.py lines 200-231
def _apply_ner(text: str, cache: Optional[AnrufAnonymisierer], nlp) -> Tuple[str, str]:
    tier = 'A'
    doc = nlp(text)
    # Offset-basiertes Replacement (Entities von hinten nach vorne um Offset-Shift zu vermeiden)
    spans = [(ent.start_char, ent.end_char, ent.label_, ent.text) for ent in doc.ents
             if ent.label_ in ('PER', 'LOC', 'ORG')]
    spans.sort(key=lambda x: x[0], reverse=True)
    for start, end, label, ent_text in spans:
        token_type = 'PERSON' if label == 'PER' else label
        is_uncertain = (len(ent_text) < 3 or not ent_text[0].isupper()
                        or any(c in ent_text for c in ['/', '\\', '@', '#']))
        if is_uncertain:
            tier = 'B'
        if cache:
            token = cache.get_or_assign_token(ent_text, token_type)
        else:
            token = f'[{token_type}_X]'
        text = text[:start] + token + text[end:]
    return (text, tier)
```
NOTE: GLiNER does NOT return char offsets (only `{'text': str, 'label': str}`). For `_apply_ner_gliner()`, use `text.find(ent_text)` to locate. Verify exact return format in Wave 0 before integrating.

# KORREKT:
from services.anonymization import extract_entities

def classify_contact(transcript_window: list, briefing_ceo_name: str,
                     current_mode: str) -> str:
    """Konsens-Voting: spaCy UND GLiNER muessen denselben Namen erkennen.
    Returns: 'target' | 'gatekeeper' | 'unknown'
    """
    text = ' '.join(transcript_window)
    entities = extract_entities(text)  # -> List[dict] with 'source' tag
    spacy_names = {e['text'].lower() for e in entities
                   if e['source'] == 'spacy' and e['type'] == 'PERSON'}
    gliner_names = {e['text'].lower() for e in entities
                    if e['source'] == 'gliner' and e['type'] == 'PERSON'}
    # Consensus: both sources must agree
    consensus_names = spacy_names & gliner_names
    if not consensus_names:
        return 'unknown'
    ceo = briefing_ceo_name.lower().strip() if briefing_ceo_name else ''
    for name in consensus_names:
        if ceo and (ceo in name or name in ceo):
            return 'target'
    return 'gatekeeper'
```
