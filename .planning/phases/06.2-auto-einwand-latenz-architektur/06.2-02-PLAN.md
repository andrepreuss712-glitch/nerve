        keyword_matchers: dict[str, EinwandKeywordMatcher] = {}  # sid -> Matcher
        keyword_matchers_lock = threading.Lock()

        def get_matcher(sid: str) -> EinwandKeywordMatcher:
            with keyword_matchers_lock:
                if sid not in keyword_matchers:
                    from services.einwand_keyword_matcher import EinwandKeywordMatcher
                    keyword_matchers[sid] = EinwandKeywordMatcher()

        def drop_matcher(sid: str) -> None:
            with keyword_matchers_lock:
                keyword_matchers.pop(sid, None)

  - id: task-2
    title: "deepgram_service.py: Interim-Branch um Keyword-Match erweitern"
    description: >
      In _make_on_message, im else-Branch (line ~100, is_final=False):

        else:
            sio.emit('transcript', {'type': 'interim', 'text': text}, room=sid)

            # BUG-10 Teil 2: Keyword-Match auf Interim-Transcript
            try:
                with ls.state_lock:
                    _muted = ls.state.get('mic_muted', False)
                if _muted:
                    return
                profile_daten = ls.get_active_profile() or {}
                einwaende = profile_daten.get('einwaende') or []
                if not einwaende:
                    return
                matcher = ls.get_matcher(sid)
                match = matcher.match_with_dedup(text, einwaende)
                if not match:
                    return

                # Slot 0 instant render via neuen Socket-Event
                _label = match.get('matched_label', '')
                _pe = match.get('profile_einwand') or {}
                sio.emit('keyword_einwand_match', {
                    'keyword': match['keyword'],
                    'typ': _label,
                    'profile_einwand': _pe,
                    'transcript_snippet': text[:200],
                }, room=sid)
                print(f"[KeywordMatch] sid={sid} keyword={match['keyword']} label={_label} text={text[:60]!r}")

                # Slot 1: parallele Haiku-Variante, mit eigenem busy_until-Guard
                _busy_until = getattr(_make_on_message, '_variant_busy_until', {})
                _now = time.monotonic()
                if _busy_until.get(sid, 0) < _now:
                    _busy_until[sid] = _now + 6
                    _make_on_message._variant_busy_until = _busy_until
                    with ls.buffer_lock:
                        _kontext = " ".join(ls.analysiert_bisher[-20:])
                    from services.claude_service import streame_auto_variante
                    sio.start_background_task(
                        streame_auto_variante,
                        text, einwaende, _kontext, sid, 1
                    )
            except Exception as _kw_err:
                print(f"[KeywordMatch] error sid={sid}: {_kw_err}")

      WICHTIG: Der busy_until-Dict muss auf der Funktion _make_on_message selbst

      per-sid).

  - id: task-3
    title: "UtteranceEnd-Handler fuer Dedup-Reset"
    description: >
      Deepgram emittet LiveTranscriptionEvents.UtteranceEnd wenn eine Sprachpause
      detected wird. Diesen Handler neu registrieren (falls nicht vorhanden):

        connection.on(LiveTranscriptionEvents.UtteranceEnd, _make_on_utterance_end(sid))

        def _make_on_utterance_end(sid):
            def on_utterance_end(self, utterance_end, **kwargs):
                try:
                    matcher = ls.get_matcher(sid)
                    matcher.reset_all()
                    print(f"[KeywordMatch] utterance-end reset sid={sid}")
                except Exception as e:
                    print(f"[KeywordMatch] utterance-end reset error sid={sid}: {e}")
            return on_utterance_end

      Hintergrund: Dedup-Window 10s waere zu grosszuegig fuer kurze Cold-Call-
      Gespraeche mit mehreren aufeinanderfolgenden Einwaenden. UtteranceEnd ist
      der praezisere Reset-Trigger — pro Satz ein neuer Versuch erlaubt.
