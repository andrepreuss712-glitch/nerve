### `static/pip-launcher.js` — BUG-1: EWB Labels render `[object Object]`

**Root cause (confirmed):** Profile editor stores einwaende as objects with shape `{kategorie, intensitaet, einwand, varianten, gegenargument, technik}` (see `templates/profile_editor.html` lines 830-838 `getEinwaende()`), NOT `{typ, name}`. The current renderer fallback `typeof e === 'string' ? e : (e.typ || e.name || String(e))` finds neither `.typ` nor `.name` on the profile-editor objects and falls through to `String(e)` → `"[object Object]"`.

**Analog:** `static/pip-launcher.js` `_renderEwbButtons` lines 972-986 (revise in place)

**Current broken code** (lines 975-981):
```javascript
var einwaende = (state.profileDaten && state.profileDaten.einwaende) ? state.profileDaten.einwaende : [];
if (!einwaende.length) { row.innerHTML = ''; return; }
var html = einwaende.slice(0, 5).map(function (e) {
  var typ = typeof e === 'string' ? e : (e.typ || e.name || String(e));  // ← BUG: falls through to String(e)
  return '<button class="pip-ewb-btn" data-typ="' + escHtml(typ) + '">' + escHtml(typ) + '</button>';
}).join('');
row.innerHTML = html;
```

**Fix pattern (same extraction style used elsewhere in app.js for legacy code):**
The legacy `renderEwbButtons` in `static/app.js:245` uses: `pd.einwaende.map(e => e.typ || e.name || e).filter(Boolean)` — also broken for profile-editor objects. The correct field priority for the wizard-stored format (list of strings) AND profile-editor format (objects with `.kategorie` / `.einwand`) is:

```javascript
var typ = typeof e === 'string'
  ? e
  : (e.kategorie || e.typ || e.name || e.einwand || '');
if (!typ) return '';  // skip rather than render [object Object]
```

Note: D-11 also requires all 5 buttons to render (no slice to 2). Current code already uses `.slice(0, 5)` — keep.

**D-11 CSS requirement** (horizontal pill row, no wrap to 3 lines): The existing `.pip-ewb-row` at `templates/base.html:228` uses `flex-wrap:wrap` — change to `flex-wrap:nowrap` + `overflow:hidden` and let pills shrink via `text-overflow:ellipsis`.

**Animation loop pattern (rAF throttled to ~15 Hz per D-16):**
```javascript
function _startMicLevelLoop() {
  var analyser = state.micAnalyser;
  if (!analyser) return;
  var buffer = new Uint8Array(analyser.frequencyBinCount);
  var lastTick = 0;
  function tick(ts) {
    if (!state.micAnalyser) return;  // stopped
    if (ts - lastTick >= 60) {  // ~16 fps
      lastTick = ts;
      analyser.getByteFrequencyData(buffer);
      // Bucket into N bars (e.g. 4)
      var N = 4;
      var bars = new Array(N);
      var perBar = Math.floor(buffer.length / N);
      for (var i = 0; i < N; i++) {
        var sum = 0;
        for (var j = 0; j < perBar; j++) sum += buffer[i*perBar+j];
        bars[i] = sum / perBar / 255;  // normalize 0-1
      }
      _updateMicBarsDom(bars);
    }
    state.micLevelRafId = requestAnimationFrame(tick);
  }
  state.micLevelRafId = requestAnimationFrame(tick);
}

function _updateMicBarsDom(bars) {
  for (var i = 0; i < bars.length; i++) {
    var el = pipEl('pip-mic-bar-' + i);
    if (el) el.style.height = (Math.max(0.15, bars[i]) * 100) + '%';
  }
}
```

**Pattern for mute (per D-15: "Stream pausieren, nicht killen"):**
Do NOT call `_stopMic` (that kills everything and emits `stop_live_session`). Instead toggle `MediaStreamTrack.enabled` — track stays alive, but produces silence:

```javascript
function _toggleMicMute() {
  if (!state.micStream) return;
  var tracks = state.micStream.getAudioTracks();
  if (!tracks.length) return;
  var nowMuted = tracks[0].enabled;  // currently enabled → will mute
  tracks.forEach(function (t) { t.enabled = !nowMuted; });
  state.micMuted = nowMuted;
  _updateMicIndicatorState();
}

function _updateMicIndicatorState() {
  var indicator = pipEl('pip-mic-indicator');
  if (!indicator) return;
  indicator.classList.toggle('pip-mic-muted', !!state.micMuted);
  // Freeze bars to 15% height when muted (D-14: grey, bars flat)
  if (state.micMuted) {
    for (var i = 0; i < 4; i++) {
      var b = pipEl('pip-mic-bar-' + i);
      if (b) b.style.height = '15%';
    }
  }
}
```
