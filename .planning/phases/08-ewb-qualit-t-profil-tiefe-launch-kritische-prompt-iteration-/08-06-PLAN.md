```sql
SELECT ftoe.prompt_version,
       COUNT(*) AS n,
       AVG(CASE WHEN oe.success = 1 THEN 1.0 ELSE 0.0 END) AS success_rate
FROM ft_objection_events ftoe
JOIN ft_call_sessions fcs ON fcs.id = ftoe.ft_session_id
JOIN objection_events oe
  ON oe.conversation_log_id = fcs.conversation_log_id
 AND oe.einwand_typ = ftoe.objection_type
WHERE oe.success IS NOT NULL
GROUP BY ftoe.prompt_version;
```

    ```python
    class EwbRating(Base):
        """Phase 08 D-30/D-35: Manuelle EWB-Quality-Ratings von Andre (solo pre-launch).

        3 binaere Sub-Kriterien pro EWB: klingt_wie_mensch, keine_halluzination, trifft_einwand.
        Quality-Score-Formel (D-27): (klingt + 2*halluzi + trifft) / 4 * 100.
        Eindeutig pro (conversation_log_id, einwand_typ_key) — 1 Rating pro EWB in einer Session.
        """
        __tablename__ = 'ewb_ratings'
        __table_args__ = (
            UniqueConstraint('conversation_log_id', 'einwand_typ_key',
                             name='uq_ewb_rating_per_conv_ewb'),
        )
        id = Column(Integer, primary_key=True)
        conversation_log_id = Column(Integer, ForeignKey('conversation_logs.id'), nullable=False)
        einwand_typ_key = Column(String(100), nullable=False)  # matched gegen ObjectionEvent.einwand_typ
        klingt_wie_mensch = Column(Boolean, nullable=False)
        keine_halluzination = Column(Boolean, nullable=False)
        trifft_einwand = Column(Boolean, nullable=False)
        rater_id = Column(Integer, ForeignKey('users.id'), nullable=False)
        rated_at = Column(DateTime, default=utcnow, nullable=False)
        # quality_score als computed — im Python-Property statt SQL-generated
        @property
        def quality_score(self) -> float:
            """D-27: (klingt + 2*halluzi + trifft) / 4 * 100 → Skala 0-100."""
            return ((int(bool(self.klingt_wie_mensch))
                     + 2 * int(bool(self.keine_halluzination))
                     + int(bool(self.trifft_einwand))) / 4.0) * 100
    ```

    ```python
    def _seed_ewb_scenarios(db=None):
        """Phase 08 D-34: Seed 3 Test-Szenarien A/B/C als System-Training-Scenarios.

        Scenario A "Easy": Standard-Einwand "Zu teuer" bei einfachem SaaS-Profil
        Scenario B "Profil-reich": Voll ausgefuelltes Profil inkl. branche_kontext + eigene_formulierungen
        Scenario C "Edge-Case": Multi-Einwand-Sequenz "Zu teuer" -> "Haben schon was"

        Idempotent via name-based existing-row-check.
        """
        from database.db import SessionLocal
        from database.models import TrainingScenario, Organisation

        scenarios = [
            {
                'name': '[P08-A] Varianz-Test Easy: Zu teuer',
                'beschreibung': 'Phase 08 Varianz-Test Scenario A. Standard-Einwand "Zu teuer" '
                                 'gegen einen einfachen SaaS-Verkauf. Testet Baseline-Konsistenz.',
                'einwaende': ['Zu teuer'],
                'schwierigkeit': 'leicht',
            },
            {
                'name': '[P08-B] Varianz-Test Profil-reich: Bedarfs-Frage',
                'beschreibung': 'Phase 08 Varianz-Test Scenario B. Voll ausgefuelltes Profil mit '
                                 'branche_kontext + eigene_formulierungen + beweise. '
                                 'Testet ob tiefer Profil-Input konsistent in Antworten greift.',
                'einwaende': ['Bedarf unklar', 'Zu teuer'],
                'schwierigkeit': 'mittel',
            },
            {
                'name': '[P08-C] Varianz-Test Edge-Case: Multi-Einwand-Sequenz',
                'beschreibung': 'Phase 08 Varianz-Test Scenario C. Multi-Einwand: "Zu teuer" '
                                 'gefolgt von "Haben schon was Aehnliches". Testet Stabilitaet '
                                 'bei Einwand-Kette.',
                'einwaende': ['Zu teuer', 'Haben schon was'],
                'schwierigkeit': 'schwer',
            },
        ]

        owns = db is None
        if owns:
            db = SessionLocal()
        try:
            # Hole erste Org (System-Scenarios sind zwar erstellt_von=NULL, brauchen aber org_id
            # wegen NOT NULL-Constraint — Pattern aus Phase 04.9)
            first_org = db.query(Organisation).order_by(Organisation.id.asc()).first()
            if not first_org:
                print("[DB] Phase 08 _seed_ewb_scenarios skipped: no Organisation yet")
                return

            import json as _json_seed
            for s in scenarios:
                exists = db.query(TrainingScenario).filter_by(name=s['name']).first()
                if exists:
                    continue
                # Payload-Format kann TrainingScenario-spezifisch variieren —
                # hier als JSON-String in 'beschreibung' + evtl separate Column.
                # Falls TrainingScenario.einwaende/schwierigkeit eigene Columns sind,
                # direkt zuweisen. Sonst in payload-JSON.
                ts = TrainingScenario(
                    name=s['name'],
                    beschreibung=s['beschreibung'],
                    org_id=first_org.id,
                    erstellt_von=None,  # System-Scenario
                )
                # Optional-Fields je nach Model-Struktur:
                if hasattr(ts, 'schwierigkeit'):
                    ts.schwierigkeit = s['schwierigkeit']
                if hasattr(ts, 'payload'):
                    ts.payload = _json_seed.dumps({'einwaende': s['einwaende']}, ensure_ascii=False)
                db.add(ts)
            db.commit()
            print("[DB] Phase 08 Seed: 3 varianz-test scenarios (A/B/C) inserted (idempotent)")
        finally:
            if owns:
                db.close()
    ```

    ```python
    def test_ab_stats_join_success_rate(db_session):
        """3-stufiger Join liefert korrekte (prompt_version, n, success_rate)."""
        _, u = _mk_user(db_session)
        # Seed ConversationLog + FtCallSession + 2 ObjectionEvents (1 success=True, 1 False)
        # + 2 FtObjectionEvents mit unterschiedlichen prompt_version
        conv = ConversationLog(user_id=u.id, org_id=u.org_id,
                                session_mode='cold_call', dauer=60)

        fcs = FtCallSession(user_id=u.id, conversation_log_id=conv.id, mode='cold_call')
        db_session.add(fcs); db_session.flush()

        # 2 ObjectionEvents
        oe1 = ObjectionEvent(user_id=u.id, org_id=u.org_id,
                              conversation_log_id=conv.id,
                              einwand_typ='Zu teuer', success=True)

                              conversation_log_id=conv.id,
                              einwand_typ='Bedarf unklar', success=False)

        # 2 FtObjectionEvents
        ftoe1 = FtObjectionEvent(ft_session_id=fcs.id, objection_type='Zu teuer',
                                  prompt_version='v1-legacy')
        ftoe2 = FtObjectionEvent(ft_session_id=fcs.id, objection_type='Bedarf unklar',
                                  prompt_version='v2-modular')
        db_session.add_all([ftoe1, ftoe2])
    ```

    ```python
    """Phase 08 — EWB-Quality Admin Blueprint.

    2 Pages:
      - /admin/ewb-quality: A/B-Stats-Card mit 3-stufigem SQL-Join (D-22)
      - /admin/ewb-rating-template: Bulk-Rating-UI fuer Andre (100 EWBs x 3 Kriterien)

    Alle Routes @superadmin_required.
    """
    from __future__ import annotations
    from flask import Blueprint, render_template, request, jsonify, abort, g
    from sqlalchemy import text
    from datetime import datetime

    from routes.auth import login_required
    from services.auth_decorators import superadmin_required
    from database.db import get_session
    from database.models import (EwbRating, ObjectionEvent, ConversationLog,
                                  PromptVersion, FtCallSession, FtObjectionEvent, User)


    admin_ewb_bp = Blueprint(
        'admin_ewb', __name__,
        url_prefix='/admin/ewb',
        template_folder='../templates/admin',
    )


    @admin_ewb_bp.route('/quality')
    @login_required
    @superadmin_required
    def ewb_quality():
        """A/B-Stats-Card: success_rate pro prompt_version + Varianz-Signal."""
        db = get_session()
        try:
            ab_rows = db.execute(text("""
                SELECT ftoe.prompt_version AS version,
                       COUNT(*) AS n,
                       AVG(CASE WHEN oe.success = 1 THEN 1.0 ELSE 0.0 END) AS success_rate
                FROM ft_objection_events ftoe
                JOIN ft_call_sessions fcs ON fcs.id = ftoe.ft_session_id
                JOIN objection_events oe
                  ON oe.conversation_log_id = fcs.conversation_log_id
                 AND oe.einwand_typ = ftoe.objection_type
                WHERE oe.success IS NOT NULL
                GROUP BY ftoe.prompt_version
                ORDER BY ftoe.prompt_version
            """)).fetchall()

            # Rating-Score-Breakdown (D-27): 80% >= 80 Gate
            rating_rows = db.query(EwbRating).all()
            scores = [r.quality_score for r in rating_rows]
            total = len(scores) or 0
            high = sum(1 for s in scores if s >= 80) if total else 0
            pct_high = (high / total * 100) if total else 0.0

            # Varianz (D-28): Range max-min ueber Session-Scoring
            # Nur wenn Training-Szenarien A/B/C verwendet wurden → range pro scenario
            # Simplification fuer Phase 08 Wave 6: global range ueber alle rated sessions
            # NOTE: D-28 spricht konzeptionell von gesamt_score — persistiert als kb_end
            # (siehe routes/training.py:726: kb_end=scoring.get('gesamt_score', 0) und
            # services/integration_engine.py:217). RESEARCH Open Question 5 = RESOLVED.
            rated_conv_ids = [r.conversation_log_id for r in rating_rows]
            if rated_conv_ids:
                score_rows = db.execute(text("""
                    SELECT COALESCE(kb_end, 0) AS s FROM conversation_logs
                    WHERE id IN :ids
                """), {'ids': tuple(rated_conv_ids) if len(rated_conv_ids) > 1
                                   else (rated_conv_ids[0], rated_conv_ids[0])}).fetchall()
                session_scores = [r[0] for r in score_rows if r[0] is not None]
                varianz_range = (max(session_scores) - min(session_scores)) if session_scores else 0
            else:
                varianz_range = 0

            return render_template('admin/ewb_quality.html',
                                    ab_rows=ab_rows,
                                    total_ratings=total,
                                    high_score_count=high,
                                    pct_high=pct_high,
                                    varianz_range=varianz_range)
        finally:
            db.close()


    @admin_ewb_bp.route('/rating-template')
    @login_required
    @superadmin_required
    def ewb_rating_template():
        """Bulk-Rating-UI: Alle bisher ungerateten (oder gerateten) ObjectionEvents."""
        db = get_session()
        try:
            # Hole alle ObjectionEvents mit conv-Daten; LEFT JOIN auf EwbRating
            q = db.execute(text("""
                SELECT oe.id AS oe_id,
                       oe.einwand_typ,
                       oe.option_gewaehlt,
                       oe.success,
                       oe.conversation_log_id AS conv_id,
                       cl.session_mode,
                       cl.created_at,
                       r.klingt_wie_mensch,
                       r.keine_halluzination,
                       r.trifft_einwand
                FROM objection_events oe
                JOIN conversation_logs cl ON cl.id = oe.conversation_log_id
                LEFT JOIN ewb_ratings r
                  ON r.conversation_log_id = oe.conversation_log_id
                 AND r.einwand_typ_key = oe.einwand_typ
                ORDER BY cl.created_at DESC, oe.id ASC
                LIMIT 200
            """)).fetchall()
            return render_template('admin/ewb_rating_template.html', events=q)
        finally:
            db.close()


    @admin_ewb_bp.post('/rating-template/<int:conv_id>/<einwand_key>/rate')
    @login_required
    @superadmin_required
    def ewb_rating_save(conv_id, einwand_key):
        """AJAX-Save fuer ein einzelnes EWB-Rating. Idempotent via UniqueConstraint."""
        data = request.get_json(silent=True) or {}
        # Validierung: alle 3 Kriterien muessen Boolean sein
        klingt = data.get('klingt_wie_mensch')
        halluzi = data.get('keine_halluzination')
        trifft = data.get('trifft_einwand')
        if any(x not in (True, False) for x in (klingt, halluzi, trifft)):
            return jsonify({'error': 'invalid_criteria'}), 400

        db = get_session()
        try:
            existing = (db.query(EwbRating)
                        .filter_by(conversation_log_id=conv_id, einwand_typ_key=einwand_key)
                        .first())
            if existing:
                existing.klingt_wie_mensch = klingt
                existing.keine_halluzination = halluzi
                existing.trifft_einwand = trifft
                existing.rater_id = g.user.id
                existing.rated_at = datetime.utcnow()
            else:
                db.add(EwbRating(
                    conversation_log_id=conv_id,
                    einwand_typ_key=einwand_key,
                    klingt_wie_mensch=klingt,
                    keine_halluzination=halluzi,
                    trifft_einwand=trifft,
                    rater_id=g.user.id,
                ))
            db.commit()
            print(f"[EWB-Rating] conv={conv_id} ewb='{einwand_key[:30]}' "
                  f"klingt={klingt} halluzi={halluzi} trifft={trifft}")
            return jsonify({'ok': True})
        finally:
            db.close()
    ```
