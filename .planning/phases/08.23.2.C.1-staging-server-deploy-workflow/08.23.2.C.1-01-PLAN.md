```bash
echo "[setup] Configuring Postgres user and databases (idempotent)..."

su - postgres -c "psql -tAc \"SELECT 1 FROM pg_database WHERE datname='nerve'\" | grep -q 1 || psql -c \"CREATE DATABASE nerve OWNER nerve_app;\""

su - postgres -c "psql -tAc \"SELECT 1 FROM pg_database WHERE datname='nerve_test'\" | grep -q 1 || psql -c \"CREATE DATABASE nerve_test OWNER nerve_app;\""

echo "[setup] Enabling nginx..."
systemctl enable nginx
systemctl start nginx || true

echo "[setup] === Setup abgeschlossen ==="
echo "[setup] Naechste Schritte laut RUNBOOK-staging.md:"
echo "[setup]   1. /etc/nerve/.env anlegen (Vorlage: .env.staging.example)"
```

**Datei 2: deploy/nginx-staging.conf**

REVIEW-HIGH-1 FIX: Das Production-Gate in deploy.sh ruft `curl https://staging.getnerve.app/api/health`
OHNE Basic-Auth-Credentials auf. Damit dieser curl-Aufruf nicht mit 401 scheitert, muss der
`/api/health`-Location-Block VOR dem `location /`-Block stehen und `auth_basic off` setzen.
Gleiches gilt fuer `/socket.io/` (WebSocket + Basic Auth vertragen sich nicht zuverlaessig).

Erstelle nginx-Konfiguration fuer staging.getnerve.app mit:
- HTTP (Port 80) → HTTPS Redirect
- HTTPS (Port 443) mit SSL-Platzhaltern fuer Let's Encrypt
- Globale HTTP-Basic-Auth (`auth_basic "NERVE Staging"; auth_basic_user_file /etc/htpasswd.nerve-staging;`) auf Server-Ebene
- `location /api/health { auth_basic off; ... }` VOR `location /` (REVIEW-HIGH-1)
- `location /socket.io/ { auth_basic off; ... }` — Basic Auth blockiert WebSocket-Upgrades (LOW-Finding)
- robots.txt-Location mit `Disallow: /` (auth_basic erbt von Server-Ebene — OK, da Crawler ohnehin geblockt)
- Gleiche proxy_pass-Struktur wie Production (127.0.0.1:8000)
- Stripe-Webhook-Location identisch zu Production

```nginx
# ── NERVE Staging nginx-Konfiguration ────────────────────────────────────────
# Kopiere nach: /etc/nginx/sites-available/nerve-staging
# Aktiviere:   ln -s /etc/nginx/sites-available/nerve-staging /etc/nginx/sites-enabled/
# SSL-Certs:  certbot --nginx -d staging.getnerve.app

server {
    listen 80;
    server_name staging.getnerve.app;
    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl http2;
    server_name staging.getnerve.app;

    ssl_certificate     /etc/letsencrypt/live/staging.getnerve.app/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/staging.getnerve.app/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    client_max_body_size 10m;

    # Staging-Schutz: HTTP-Basic-Auth auf allen Routen (Req-8)
    # Ausnahmen: /api/health und /socket.io/ — siehe location-Bloecke unten
    auth_basic "NERVE Staging";
    auth_basic_user_file /etc/htpasswd.nerve-staging;

    # Suchmaschinen-Sperrung (Req-8) — erbt auth_basic von Server-Ebene
    location = /robots.txt {
        add_header Content-Type text/plain;
        return 200 "User-agent: *\nDisallow: /\n";
    }

    # REVIEW-HIGH-1 FIX: Health-Endpoint ohne Basic-Auth
    # deploy.sh Production-Gate ruft diesen Endpoint via curl ohne Credentials auf.
    # auth_basic off ueberschreibt die Server-Ebene fuer diese Location.
    location = /api/health {
        auth_basic off;
        proxy_pass http://127.0.0.1:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # WebSocket support for Socket.IO
    # auth_basic off: Basic Auth blockiert WebSocket-Upgrade-Handshake (LOW-Finding)
    location /socket.io/ {
        auth_basic off;
        proxy_pass http://127.0.0.1:8000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 3600s;
        proxy_send_timeout 3600s;
    }

    # Stripe webhook — raw body required for signature verification
    location /payments/webhook {
        proxy_request_buffering off;
        proxy_pass http://127.0.0.1:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        client_max_body_size 1m;
    }

    # All other requests — Basic-Auth greift (Server-Ebene)
    location / {
        proxy_pass http://127.0.0.1:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
```
