```js
  { begriff: 'verlieren',   alternative: 'absichern' },
  { begriff: 'Konkurrenz',  alternative: 'Mitbewerber' },

// Helper: case-insensitive lookup of a default Alternative by Begriff.
function findDefaultAlternative(begriff) {
  if (!begriff) return '';
  var needle = begriff.trim().toLowerCase();
  for (var i = 0; i < TABU_DEFAULT_PAIRS.length; i++) {
    if (TABU_DEFAULT_PAIRS[i].begriff.toLowerCase() === needle) {
      return TABU_DEFAULT_PAIRS[i].alternative;
    }
  }
  return '';
}
```

Then, in the `// ── Tabu-Begriffe 2-column UI` section, add a `mergeDefaultPairs()` function AFTER `syncTabuHidden()` and BEFORE `loadTabu()`:

```js
// ── mergeDefaultPairs: dedupe-merge 13 defaults into existing in-memory rows ──
// Stats: added (new row), completed (empty alt filled with default), already (begriff matched and alt already had user value)
function mergeDefaultPairs() {
  if (!tabuRowsContainer) return;
  var rows = tabuRowsContainer.querySelectorAll('.tabu-row');

  // Build index of existing begriffe (case-insensitive) → row element
  var existing = {};
  rows.forEach(function (row) {
    var bEl = row.querySelector('.tabu-begriff');
    if (!bEl) return;
    var key = (bEl.value || '').trim().toLowerCase();
    if (key) existing[key] = row;
  });

  var added = 0, completed = 0, already = 0;

  TABU_DEFAULT_PAIRS.forEach(function (pair) {
    var key = pair.begriff.toLowerCase();
    var row = existing[key];
    if (row) {
      var altEl = row.querySelector('.tabu-alternative');
      if (!altEl) return;
      var currentAlt = (altEl.value || '').trim();
      if (!currentAlt && pair.alternative) {
        altEl.value = pair.alternative;
        completed++;
      } else {
        already++;
      }
    } else {
      renderTabuRow({ begriff: pair.begriff, alternative: pair.alternative });
      added++;
    }
  });

  syncTabuHidden();
  validateTabuRows();

  // Feedback message
  var fb = document.getElementById('tabu-seed-feedback');
  if (fb) {
    var parts = [];
    if (added)     parts.push(added + ' hinzugefügt');
    if (completed) parts.push(completed + ' ergänzt');
    if (already)   parts.push(already + ' schon vollständig');
    fb.textContent = parts.length ? parts.join(', ') : '0 Änderungen';
    // Clear after 6s
    if (fb._fbTimer) clearTimeout(fb._fbTimer);
    fb._fbTimer = setTimeout(function () { fb.textContent = ''; }, 6000);
  }
}
```

Finally, wire the button INSIDE the existing `init()` function (alongside `tabuAddBtn` handling — actually the add-btn handler is registered at module scope, not in init — so register the seed-btn handler at the SAME module scope, just below the existing `if (tabuAddBtn) { ... }` block):

**Atomic commit 2 — placeholder-on-match suggestion.**

Extend `renderTabuRow(pair)` in `static/profile_editor.js` so that whenever a row is rendered, if the Begriff matches a default (case-insensitive, exact) AND the Alternative input value is empty, the Alternative input's HTML `placeholder` attribute displays `Vorschlag: <default>`. Otherwise the existing generic placeholder `Alternative (stattdessen nutzen)` remains.

Also: when the user edits the Begriff input (`input` event on `.tabu-begriff`), re-evaluate the placeholder on the same row's Alternative input — so typing "kosten" into a new blank row immediately updates the hint to `Vorschlag: Investition`.

### Concrete diff in renderTabuRow

1. After the `alternativeInput` is constructed (currently placeholder `'Alternative (stattdessen nutzen)'`), add a helper local:

```js
function applyPlaceholderSuggestion() {
  var currentBegriff = begriffInput.value || '';
  var currentAlt = alternativeInput.value || '';
  if (currentAlt.trim()) {
    // user has own value — keep generic placeholder (not visible anyway)
    alternativeInput.placeholder = 'Alternative (stattdessen nutzen)';
    return;
  }
  var suggestion = findDefaultAlternative(currentBegriff);
  if (suggestion) {
    alternativeInput.placeholder = 'Vorschlag: ' + suggestion;
  } else {
    alternativeInput.placeholder = 'Alternative (stattdessen nutzen)';
  }
}
```

2. Call `applyPlaceholderSuggestion()` immediately after creating both inputs (before appendChild) so the initial render is correct.

3. In the existing `[begriffInput, alternativeInput].forEach(function (el) { el.addEventListener('input', ...) })` block, ALSO call `applyPlaceholderSuggestion()` on every input event — BEFORE the existing `validateTabuRows()` / `syncTabuHidden()` calls. Small edit:

```js
[begriffInput, alternativeInput].forEach(function (el) {
  el.addEventListener('input', function () {
    applyPlaceholderSuggestion();
    validateTabuRows();
    syncTabuHidden();
  });
});
```

### Constraints / do NOT
- Do NOT auto-write the placeholder value into `alternativeInput.value`. The placeholder is an HTML attribute — the value stays empty until the user types. If the user wants the suggestion, they type it or use the Fix A button.
- Do NOT trigger the placeholder logic on rows whose Alternative is non-empty (keeps the UI quiet).
- Do NOT add the placeholder for partial / substring matches — use the exact case-insensitive match from `findDefaultAlternative()` (already defined in Task 1).
